# Real team tests are opt-in by supplying their source; no placeholder PASS.
if(NOT EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/integration_tb.cpp")
  message(STATUS "h264_coding_reconstruction_integration_tb: not registered (integration_tb.cpp not delivered)")
  return()
endif()
set(H264_CODING_RECONSTRUCTION_MODULE_DIR "" CACHE PATH "Optional directory exporting real team module targets")
set(H264_CODING_RECONSTRUCTION_TEST_LIBRARIES "" CACHE STRING "Semicolon-separated real module targets")
if(H264_CODING_RECONSTRUCTION_MODULE_DIR)
  add_subdirectory("${H264_CODING_RECONSTRUCTION_MODULE_DIR}" "${CMAKE_CURRENT_BINARY_DIR}/modules")
endif()
if(NOT H264_CODING_RECONSTRUCTION_TEST_LIBRARIES)
  message(FATAL_ERROR "Set H264_CODING_RECONSTRUCTION_TEST_LIBRARIES to real module targets")
endif()
foreach(lib IN LISTS H264_CODING_RECONSTRUCTION_TEST_LIBRARIES)
  if(NOT TARGET ${lib})
    message(FATAL_ERROR "Missing team target: ${lib}")
  endif()
endforeach()
add_executable(h264_coding_reconstruction_integration_tb integration_tb.cpp)
target_link_libraries(h264_coding_reconstruction_integration_tb PRIVATE h264_top h264_platform ${H264_CODING_RECONSTRUCTION_TEST_LIBRARIES})
add_test(NAME h264_coding_reconstruction_integration_tb COMMAND h264_coding_reconstruction_integration_tb)
set_tests_properties(h264_coding_reconstruction_integration_tb PROPERTIES TIMEOUT 60 LABELS "h264;team;coding_reconstruction")
