cmake_minimum_required(VERSION 3.21)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
  project(h264_vp VERSION 0.1.0 LANGUAGES CXX)
  set(CMAKE_CXX_STANDARD 17)
  set(CMAKE_CXX_STANDARD_REQUIRED ON)
  set(CMAKE_CXX_EXTENSIONS OFF)
  set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/bin")
  if(SYSTEMC_INCLUDE_DIR AND SYSTEMC_LIBRARY)
    add_library(SystemC::systemc UNKNOWN IMPORTED GLOBAL)
    set_target_properties(SystemC::systemc PROPERTIES
      IMPORTED_LOCATION "${SYSTEMC_LIBRARY}"
      INTERFACE_INCLUDE_DIRECTORIES "${SYSTEMC_INCLUDE_DIR}")
  elseif(SYSTEMC_SOURCE_DIR)
    set(CMAKE_EXPORT_PACKAGE_REGISTRY OFF)
    set(CMAKE_EXPORT_NO_PACKAGE_REGISTRY ON)
    set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
    set(BUILD_SOURCE_DOCUMENTATION OFF CACHE BOOL "" FORCE)
    add_subdirectory("${SYSTEMC_SOURCE_DIR}" "${CMAKE_CURRENT_BINARY_DIR}/systemc" EXCLUDE_FROM_ALL)
  else()
    find_package(SystemCLanguage 2.3.4 EXACT CONFIG REQUIRED)
  endif()
  enable_testing()
endif()
# Integration package. Public target and include names remain stable for team adapters.
if(NOT TARGET SystemC::systemc)
  message(FATAL_ERROR "The parent must provide SystemC::systemc (2.3.4)")
endif()
set(_h264_standalone OFF)
if(CMAKE_SOURCE_DIR STREQUAL PROJECT_SOURCE_DIR AND PROJECT_NAME STREQUAL "h264_vp")
  set(_h264_standalone ON)
endif()
set(_h264_tests_default ${_h264_standalone})
if(DEFINED CDC_BUILD_TESTS)
  set(_h264_tests_default ${CDC_BUILD_TESTS})
endif()
option(H264_BUILD_TESTS "Build H264 component regression" ${_h264_tests_default})
option(H264_BUILD_EXAMPLES "Build standalone H264 demo" ${_h264_standalone})
option(H264_BUILD_STUBS "Build explicit non-codec pipeline fixture" ${_h264_standalone})
option(H264_BUILD_CDC_TESTS "Test against an external CDC-VP FX1 source tree" OFF)
set(H264_CDC_VP_ROOT "" CACHE PATH "Read-only CDC-VP source tree for integration tests")
add_subdirectory(interfaces)
add_subdirectory(synchronization)
add_subdirectory(control)
add_subdirectory(axi_dma)
add_subdirectory(top)
add_library(cdc::components::h264_tlm ALIAS h264_top)
if(H264_BUILD_STUBS OR H264_BUILD_TESTS OR H264_BUILD_EXAMPLES OR H264_BUILD_CDC_TESTS)
  if(H264_BUILD_TESTS OR H264_BUILD_CDC_TESTS)
    enable_testing()
  endif()
  add_subdirectory(tests)
endif()
include(cmake/install.cmake)
