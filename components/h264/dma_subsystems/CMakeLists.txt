cmake_minimum_required(VERSION 3.16)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
  project(h264_dma_vp LANGUAGES CXX)
  include(CTest)
  option(H264_BUILD_TLM "Build SystemC/TLM adapter and shared-bus platform" OFF)
  set(H264_BUILD_TESTS ${BUILD_TESTING})
else()
  set(H264_BUILD_TLM ON)
  if(NOT DEFINED H264_BUILD_TESTS)
    set(H264_BUILD_TESTS ${CDC_BUILD_TESTS})
  endif()
endif()
add_library(h264_dma INTERFACE)
add_library(cdc::components::h264_dma ALIAS h264_dma)
foreach(dir common dma mem)
  target_include_directories(h264_dma INTERFACE
    $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/${dir}>
    $<INSTALL_INTERFACE:include/h264/dma/${dir}>)
endforeach()
target_compile_features(h264_dma INTERFACE cxx_std_17)
if(H264_BUILD_TESTS)
  add_executable(h264_dma_tests tb/vp_tests.cpp tb/spec_review_tests.cpp)
  target_link_libraries(h264_dma_tests PRIVATE h264_dma)
  target_compile_definitions(h264_dma_tests PRIVATE H264_WITH_REVIEW H264_COMBINED_TESTS)
  if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
    target_compile_options(h264_dma_tests PRIVATE -Wall -Wextra -Wpedantic)
  endif()
  add_test(NAME h264_dma_contracts COMMAND h264_dma_tests)
endif()
if(H264_BUILD_TLM)
  if(NOT TARGET SystemC::systemc)
    set(SYSTEMC_HOME "/opt/systemc-2.3.4" CACHE PATH "SystemC installation prefix")
    find_path(SYSTEMC_INCLUDE_DIR NAMES systemc HINTS "${SYSTEMC_HOME}/include")
    find_library(SYSTEMC_LIBRARY NAMES systemc HINTS "${SYSTEMC_HOME}/lib" "${SYSTEMC_HOME}/lib64")
    if(NOT SYSTEMC_INCLUDE_DIR OR NOT SYSTEMC_LIBRARY)
      message(FATAL_ERROR "Set SYSTEMC_HOME to an installed SystemC prefix")
    endif()
    add_library(SystemC::systemc UNKNOWN IMPORTED GLOBAL)
    set_target_properties(SystemC::systemc PROPERTIES IMPORTED_LOCATION "${SYSTEMC_LIBRARY}"
      INTERFACE_INCLUDE_DIRECTORIES "${SYSTEMC_INCLUDE_DIR}")
  endif()
  add_library(h264_dma_tlm STATIC tlm/src/h264_dma_tlm.cpp)
  add_library(cdc::components::h264_dma_tlm ALIAS h264_dma_tlm)
  target_include_directories(h264_dma_tlm PUBLIC
    $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/tlm/include>
    $<INSTALL_INTERFACE:include/h264/dma>)
  target_link_libraries(h264_dma_tlm PUBLIC h264_dma SystemC::systemc)
  if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
    target_compile_options(h264_dma_tlm PRIVATE -Wall -Wextra -Wpedantic)
  endif()
  if(H264_BUILD_TESTS)
      add_executable(h264_dma_platform tlm/tests/h264_dma_platform.cpp
        ../../bus_router/src/bus_router.cpp ../../memory_tlm/src/memory_tlm.cpp)
      target_include_directories(h264_dma_platform PRIVATE ../../bus_router/include
        ../../memory_tlm/include ../../common/include)
      target_link_libraries(h264_dma_platform PRIVATE h264_dma_tlm)
      add_test(NAME h264_dma_shared_bus COMMAND h264_dma_platform)
      add_test(NAME h264_dma_shared_bus_64 COMMAND h264_dma_platform 64)
      add_test(NAME h264_dma_shared_bus_128 COMMAND h264_dma_platform 128)
      add_test(NAME h264_dma_response_timeout COMMAND h264_dma_platform --timeout)
      set_tests_properties(h264_dma_shared_bus h264_dma_shared_bus_64
        h264_dma_shared_bus_128 h264_dma_response_timeout PROPERTIES TIMEOUT 30)
  endif()
  if(NOT TARGET h264_control AND NOT CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
    install(TARGETS h264_dma h264_dma_tlm EXPORT cdc-components-targets ARCHIVE DESTINATION lib)
    install(DIRECTORY common dma mem DESTINATION include/h264/dma FILES_MATCHING PATTERN "*.h")
    install(FILES tlm/include/h264_dma_tlm.h DESTINATION include/h264/dma)
  endif()
endif()
