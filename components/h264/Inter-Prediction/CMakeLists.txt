cmake_minimum_required(VERSION 3.21)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
  project(h264_inter_vp LANGUAGES CXX)
  set(CMAKE_CXX_STANDARD 17)
  set(CMAKE_CXX_STANDARD_REQUIRED ON)
  set(CMAKE_CXX_EXTENSIONS OFF)
  include(CTest)
  set(H264_BUILD_TESTS ${BUILD_TESTING})
endif()
option(H264_INTER_BUILD_TLM "Build SystemC/TLM Inter VP" ON)
option(H264_INTER_BUILD_DMA_TESTS "Test with sibling DMA subsystem models" ON)
add_library(h264_inter_core STATIC
  src/cache/reference_cache.cpp src/ime/sad_tree.cpp
  src/fme/interpolation.cpp src/fme/search.cpp
  src/mc/motion_compensation.cpp src/eei/motion_syntax.cpp)
add_library(cdc::components::h264_inter_core ALIAS h264_inter_core)
target_compile_features(h264_inter_core PUBLIC cxx_std_17)
target_include_directories(h264_inter_core PUBLIC
  $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include>
  $<INSTALL_INTERFACE:include>)
if(H264_INTER_BUILD_TLM)
  if(NOT TARGET SystemC::systemc)
    if(SYSTEMC_INCLUDE_DIR AND SYSTEMC_LIBRARY)
      add_library(SystemC::systemc UNKNOWN IMPORTED GLOBAL)
      set_target_properties(SystemC::systemc PROPERTIES IMPORTED_LOCATION "${SYSTEMC_LIBRARY}"
        INTERFACE_INCLUDE_DIRECTORIES "${SYSTEMC_INCLUDE_DIR}")
    elseif(SYSTEMC_SOURCE_DIR)
      set(CMAKE_CXX_STANDARD 17)
      set(CMAKE_CXX_STANDARD_REQUIRED ON)
      set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
      add_subdirectory("${SYSTEMC_SOURCE_DIR}" "${CMAKE_CURRENT_BINARY_DIR}/systemc" EXCLUDE_FROM_ALL)
    else()
      find_package(SystemCLanguage 2.3.4 EXACT CONFIG REQUIRED)
    endif()
  endif()
  # Reuse existing epoch contracts; no parent/top modifications.
  if(NOT TARGET h264_shared)
    add_subdirectory(../interfaces "${CMAKE_CURRENT_BINARY_DIR}/interfaces")
  endif()
  if(NOT TARGET h264_sync)
    add_subdirectory(../synchronization "${CMAKE_CURRENT_BINARY_DIR}/synchronization")
  endif()
  add_library(h264_inter_tlm STATIC src/ipc/inter_tlm.cpp)
  add_library(cdc::components::h264_inter_tlm ALIAS h264_inter_tlm)
  target_link_libraries(h264_inter_tlm PUBLIC h264_inter_core h264_sync)
  target_include_directories(h264_inter_tlm PUBLIC
    $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include>
    $<INSTALL_INTERFACE:include>)
endif()
foreach(lib h264_inter_core h264_inter_tlm)
  if(TARGET ${lib} AND CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
    target_compile_options(${lib} PRIVATE -Wall -Wextra -Wpedantic)
  endif()
endforeach()
if(H264_BUILD_TESTS)
  add_executable(h264_inter_core_tests tests/core/core_tests.cpp)
  target_link_libraries(h264_inter_core_tests PRIVATE h264_inter_core)
  add_test(NAME h264_inter_sad_cache_search COMMAND h264_inter_core_tests)
  add_executable(h264_inter_interpolation_tests tests/core/interpolation_tests.cpp)
  target_link_libraries(h264_inter_interpolation_tests PRIVATE h264_inter_core)
  add_test(NAME h264_inter_interpolation_oracle COMMAND h264_inter_interpolation_tests)
  add_executable(h264_inter_limits_tests tests/core/limits_tests.cpp)
  target_link_libraries(h264_inter_limits_tests PRIVATE h264_inter_core)
  add_test(NAME h264_inter_limits COMMAND h264_inter_limits_tests)
  set_tests_properties(h264_inter_limits PROPERTIES TIMEOUT 60)
  if(H264_INTER_BUILD_DMA_TESTS)
    if(NOT EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/../dma_subsystems/dma/sw_dma.h")
      message(FATAL_ERROR "SW DMA tests need ../dma_subsystems; set H264_INTER_BUILD_DMA_TESTS=OFF for isolated core/contract tests")
    endif()
    add_executable(h264_inter_sw_dma_tests tests/integration/sw_dma_tests.cpp)
    target_link_libraries(h264_inter_sw_dma_tests PRIVATE h264_inter_core)
    target_include_directories(h264_inter_sw_dma_tests PRIVATE
      ../dma_subsystems/common ../dma_subsystems/dma ../dma_subsystems/mem)
    foreach(width 32 64 128)
      add_test(NAME h264_inter_sw_dma_${width} COMMAND h264_inter_sw_dma_tests ${width})
      set_tests_properties(h264_inter_sw_dma_${width} PROPERTIES TIMEOUT 30)
    endforeach()
  endif()
  if(H264_INTER_BUILD_TLM)
    add_executable(h264_inter_contract_tests tests/contract/tlm_tests.cpp)
    target_link_libraries(h264_inter_contract_tests PRIVATE h264_inter_tlm)
    foreach(scenario normal zero shared local disable retag incoming stale partitions partitions-zero boundaries boundary-zero)
      add_test(NAME h264_inter_contract_${scenario} COMMAND h264_inter_contract_tests ${scenario})
      set_tests_properties(h264_inter_contract_${scenario} PROPERTIES TIMEOUT 30)
    endforeach()
  endif()
endif()
