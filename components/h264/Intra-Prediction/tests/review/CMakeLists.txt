cmake_minimum_required(VERSION 3.21)
project(intra_review LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
enable_testing()
add_library(SystemC::systemc UNKNOWN IMPORTED GLOBAL)
set_target_properties(SystemC::systemc PROPERTIES
  IMPORTED_LOCATION "${SYSTEMC_LIBRARY}"
  INTERFACE_INCLUDE_DIRECTORIES "${SYSTEMC_INCLUDE_DIR}")
set(H264_BUILD_TESTS OFF)
add_subdirectory(../.. intra)
add_executable(intra_review_core core_review.cpp)
target_link_libraries(intra_review_core PRIVATE h264_intra_core)
add_test(NAME intra_review_core COMMAND intra_review_core)
add_executable(intra_review_epoch epoch_review.cpp)
target_link_libraries(intra_review_epoch PRIVATE h264_intra_tlm)
add_test(NAME intra_review_epoch_incoming COMMAND intra_review_epoch incoming)
add_test(NAME intra_review_epoch_wakeup COMMAND intra_review_epoch wakeup)
set_tests_properties(intra_review_epoch_incoming intra_review_epoch_wakeup PROPERTIES TIMEOUT 10)
