cmake_minimum_required(VERSION 3.21)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
  project(h264_intra_vp LANGUAGES CXX)
  set(CMAKE_CXX_STANDARD 17)
  set(CMAKE_CXX_STANDARD_REQUIRED ON)
  set(CMAKE_CXX_EXTENSIONS OFF)
  include(CTest)
  set(H264_BUILD_TESTS ${BUILD_TESTING})
  option(H264_INTRA_BUILD_TLM "Build SystemC/TLM model" ON)
else()
  set(H264_INTRA_BUILD_TLM ON)
endif()
add_library(h264_intra_core STATIC src/intra_core.cpp)
add_library(cdc::components::h264_intra_core ALIAS h264_intra_core)
target_compile_features(h264_intra_core PUBLIC cxx_std_17)
target_include_directories(h264_intra_core PUBLIC
  $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include>
  $<INSTALL_INTERFACE:include>)
if(H264_INTRA_BUILD_TLM)
  if(NOT TARGET SystemC::systemc)
    if(SYSTEMC_INCLUDE_DIR AND SYSTEMC_LIBRARY)
      add_library(SystemC::systemc UNKNOWN IMPORTED GLOBAL)
      set_target_properties(SystemC::systemc PROPERTIES IMPORTED_LOCATION "${SYSTEMC_LIBRARY}"
        INTERFACE_INCLUDE_DIRECTORIES "${SYSTEMC_INCLUDE_DIR}")
    elseif(SYSTEMC_SOURCE_DIR)
      set(CMAKE_CXX_STANDARD 17)
      set(CMAKE_CXX_STANDARD_REQUIRED ON)
      set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
      add_subdirectory("${SYSTEMC_SOURCE_DIR}" "${CMAKE_CURRENT_BINARY_DIR}/systemc" EXCLUDE_FROM_ALL)
    else()
      find_package(SystemCLanguage 2.3.4 EXACT CONFIG REQUIRED)
    endif()
  endif()
  # Standalone reuse of the existing epoch contract, without the encoder top.
  if(NOT TARGET h264_sync)
    add_subdirectory(../interfaces "${CMAKE_CURRENT_BINARY_DIR}/interfaces")
    add_subdirectory(../synchronization "${CMAKE_CURRENT_BINARY_DIR}/synchronization")
  endif()
  add_library(h264_intra_tlm STATIC src/intra_tlm.cpp)
  add_library(cdc::components::h264_intra_tlm ALIAS h264_intra_tlm)
  target_link_libraries(h264_intra_tlm PUBLIC h264_intra_core h264_sync)
  target_include_directories(h264_intra_tlm PUBLIC
    $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include>
    $<INSTALL_INTERFACE:include>)
endif()
foreach(lib h264_intra_core h264_intra_tlm)
  if(TARGET ${lib} AND CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
    target_compile_options(${lib} PRIVATE -Wall -Wextra -Wpedantic)
  endif()
endforeach()
if(H264_BUILD_TESTS)
  add_executable(h264_intra_tests tests/core_tests.cpp)
  target_link_libraries(h264_intra_tests PRIVATE h264_intra_core)
  add_test(NAME h264_intra_algorithms COMMAND h264_intra_tests)
  add_executable(h264_intra_review_core tests/review/core_review.cpp)
  target_link_libraries(h264_intra_review_core PRIVATE h264_intra_core)
  add_test(NAME h264_intra_core_regression COMMAND h264_intra_review_core)
  if(H264_INTRA_BUILD_TLM)
    add_executable(h264_intra_tlm_tests tests/tlm_tests.cpp)
    target_link_libraries(h264_intra_tlm_tests PRIVATE h264_intra_tlm)
    add_test(NAME h264_intra_tlm_contract COMMAND h264_intra_tlm_tests)
    set_tests_properties(h264_intra_tlm_contract PROPERTIES TIMEOUT 30)
    add_executable(h264_intra_epoch_tests tests/review/epoch_review.cpp)
    target_link_libraries(h264_intra_epoch_tests PRIVATE h264_intra_tlm)
    add_test(NAME h264_intra_epoch_incoming COMMAND h264_intra_epoch_tests incoming)
    add_test(NAME h264_intra_epoch_wakeup COMMAND h264_intra_epoch_tests wakeup)
    set_tests_properties(h264_intra_epoch_incoming h264_intra_epoch_wakeup PROPERTIES TIMEOUT 10)
    add_executable(h264_intra_reset_tests tests/reset_tests.cpp)
    target_link_libraries(h264_intra_reset_tests PRIVATE h264_intra_tlm)
    add_test(NAME h264_intra_reset_matrix COMMAND h264_intra_reset_tests)
    set_tests_properties(h264_intra_reset_matrix PROPERTIES TIMEOUT 30)
    add_executable(h264_intra_tlm_regression tests/tlm_regression.cpp)
    target_link_libraries(h264_intra_tlm_regression PRIVATE h264_intra_tlm)
    add_test(NAME h264_intra_tlm_regression COMMAND h264_intra_tlm_regression)
    add_test(NAME h264_intra_tlm_zero_latency COMMAND h264_intra_tlm_regression zero)
    set_tests_properties(h264_intra_tlm_regression h264_intra_tlm_zero_latency PROPERTIES TIMEOUT 30)
    add_executable(h264_intra_reset_boundary_tests tests/reset_boundary_tests.cpp)
    target_link_libraries(h264_intra_reset_boundary_tests PRIVATE h264_intra_tlm)
    add_test(NAME h264_intra_reset_boundaries COMMAND h264_intra_reset_boundary_tests)
    add_test(NAME h264_intra_reset_boundaries_zero COMMAND h264_intra_reset_boundary_tests zero)
    set_tests_properties(h264_intra_reset_boundaries h264_intra_reset_boundaries_zero PROPERTIES TIMEOUT 30)
  endif()
endif()
