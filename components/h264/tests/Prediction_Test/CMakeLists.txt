# Only test dependencies; do not import the duplicate h264_dma target.
function(h264_prediction_dependencies)
  set(H264_BUILD_TESTS OFF)
  set(H264_INTER_BUILD_DMA_TESTS OFF)
  if(NOT TARGET h264_intra_tlm)
    add_subdirectory("${CMAKE_CURRENT_SOURCE_DIR}/../../Intra-Prediction" "${CMAKE_CURRENT_BINARY_DIR}/intra")
  endif()
  if(NOT TARGET h264_inter_tlm)
    add_subdirectory("${CMAKE_CURRENT_SOURCE_DIR}/../../Inter-Prediction" "${CMAKE_CURRENT_BINARY_DIR}/inter")
  endif()
endfunction()
h264_prediction_dependencies()
foreach(stage dma_clients cmb_intra sw_inter control_prediction)
  add_executable(h264_prediction_${stage}_tb ${stage}_tb.cpp)
  target_compile_features(h264_prediction_${stage}_tb PRIVATE cxx_std_17)
  target_include_directories(h264_prediction_${stage}_tb PRIVATE ../../dma_subsystems/common ../../dma_subsystems/dma)
  target_link_libraries(h264_prediction_${stage}_tb PRIVATE h264_top h264_platform h264_intra_tlm h264_inter_tlm)
  foreach(width 32 64 128)
    add_test(NAME h264_prediction_${stage}_${width} COMMAND h264_prediction_${stage}_tb ${width})
    set_tests_properties(h264_prediction_${stage}_${width} PROPERTIES TIMEOUT 30 LABELS "prediction;integration;fixture-backend")
  endforeach()
endforeach()
foreach(width 32 64 128)
  add_test(NAME h264_prediction_control_fault_${width} COMMAND h264_prediction_control_prediction_tb ${width} fault)
  set_tests_properties(h264_prediction_control_fault_${width} PROPERTIES TIMEOUT 30 LABELS "prediction;integration;fixture-backend")
endforeach()
