cmake_minimum_required(VERSION 3.21)
project(bus_installed_consumer LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
if(DEFINED ENV{SYSTEMC_HOME})
  list(PREPEND CMAKE_PREFIX_PATH "$ENV{SYSTEMC_HOME}")
endif()
find_package(SystemCLanguage CONFIG REQUIRED)
if(NOT CDC_COMPONENTS_PREFIX)
  message(FATAL_ERROR "Set CDC_COMPONENTS_PREFIX to the test installation prefix")
endif()
include("${CDC_COMPONENTS_PREFIX}/lib/cmake/cdc-components/cdc-components-targets.cmake")
add_executable(bus_installed_consumer main.cpp)
target_link_libraries(bus_installed_consumer PRIVATE cdc::components::bus_system)
enable_testing()
add_test(NAME installed_bus_consumer COMMAND bus_installed_consumer)
