cmake_minimum_required(VERSION 3.21)
project(bus_cdc_integration LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT CDC_VP_SOURCE_DIR)
  message(FATAL_ERROR "Set CDC_VP_SOURCE_DIR to the CDC-VP source checkout")
endif()
if(SYSTEMC_HOME)
  list(PREPEND CMAKE_PREFIX_PATH "${SYSTEMC_HOME}")
elseif(DEFINED ENV{SYSTEMC_HOME})
  list(PREPEND CMAKE_PREFIX_PATH "$ENV{SYSTEMC_HOME}")
endif()
find_package(SystemCLanguage CONFIG REQUIRED)
set(CDC_BUILD_TESTS OFF)
set(BUS_SYSTEM_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(BUS_SYSTEM_USE_CDC_EXPORT ON)
add_subdirectory(../.. bus)
if(NOT PROJECT_NAME STREQUAL "bus_cdc_integration")
  message(FATAL_ERROR "The bus must preserve its parent project")
endif()
add_subdirectory("${CDC_VP_SOURCE_DIR}/components/memory_tlm" memory_tlm)
add_executable(cdc_bus_integration integration.cpp)
target_include_directories(cdc_bus_integration PRIVATE "${CDC_VP_SOURCE_DIR}/cpu_models/include")
target_link_libraries(cdc_bus_integration PRIVATE cdc::components::bus_system cdc::components::memory_tlm)
enable_testing()
add_test(NAME cdc_bus_integration COMMAND cdc_bus_integration)
set_tests_properties(cdc_bus_integration PROPERTIES TIMEOUT 30)
# Exercise the same shared export set as the CDC-VP components parent.
install(EXPORT cdc-components-targets NAMESPACE cdc::components:: DESTINATION lib/cmake/cdc-components)
