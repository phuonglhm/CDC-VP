cmake_minimum_required(VERSION 3.21)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
    project(fx1_ethernet VERSION 1.0.6 LANGUAGES C CXX)
    include(CTest)
    set(_eth_tests_default ${BUILD_TESTING})
    set(_eth_examples_default ON)
elseif(DEFINED CDC_BUILD_TESTS)
    set(_eth_tests_default ${CDC_BUILD_TESTS})
    set(_eth_examples_default OFF)
else()
    set(_eth_tests_default OFF)
    set(_eth_examples_default OFF)
endif()
option(ICSL_ETH_VP_TESTS "Build Ethernet regression tests" ${_eth_tests_default})
option(FX1_ETH_BUILD_EXAMPLES "Build the standalone two-node example" ${_eth_examples_default})
option(ICSL_ETH_VP_CDC "Build the legacy Bremen RV32 example (not fx1_soc)" OFF)
set(CDC_VP_ROOT "" CACHE PATH "CDC-VP checkout for the legacy RV32 example")
set(ICSL_ETH_VP "${CMAKE_CURRENT_SOURCE_DIR}")

if(NOT TARGET SystemC::systemc)
    set(SYSTEMC_HOME "$ENV{SYSTEMC_HOME}" CACHE PATH "SystemC installation prefix")
    find_path(SYSTEMC_INCLUDE_DIR NAMES systemc HINTS "${SYSTEMC_HOME}/include")
    find_library(SYSTEMC_LIBRARY NAMES systemc HINTS "${SYSTEMC_HOME}/lib64" "${SYSTEMC_HOME}/lib" "${SYSTEMC_HOME}/lib-mingw64")
    if(NOT SYSTEMC_INCLUDE_DIR OR NOT SYSTEMC_LIBRARY)
        message(FATAL_ERROR "SystemC not found: set SYSTEMC_HOME")
    endif()
    find_package(Threads REQUIRED)
    add_library(SystemC::systemc UNKNOWN IMPORTED GLOBAL)
    set_target_properties(SystemC::systemc PROPERTIES IMPORTED_LOCATION "${SYSTEMC_LIBRARY}"
        INTERFACE_INCLUDE_DIRECTORIES "${SYSTEMC_INCLUDE_DIR}"
        INTERFACE_LINK_LIBRARIES Threads::Threads)
endif()

add_library(eth_core STATIC src/core/eth_core.cpp)
target_include_directories(eth_core PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}/include")
target_compile_features(eth_core PUBLIC cxx_std_17)
add_library(eth_mac_tlm STATIC src/csr_file.cpp src/eth_mac_tlm.cpp src/eth_mac_dma.cpp src/eth_mac_datapath.cpp)
target_include_directories(eth_mac_tlm PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}/include" "${CMAKE_CURRENT_SOURCE_DIR}/driver/include" PRIVATE src)
target_compile_features(eth_mac_tlm PUBLIC cxx_std_17)
target_link_libraries(eth_mac_tlm PUBLIC eth_core SystemC::systemc)
add_library(cdc::components::fx1_ethernet ALIAS eth_mac_tlm)
add_library(eth_phy_tlm INTERFACE)
target_include_directories(eth_phy_tlm INTERFACE "${CMAKE_CURRENT_SOURCE_DIR}/include")
target_compile_features(eth_phy_tlm INTERFACE cxx_std_17)
target_link_libraries(eth_phy_tlm INTERFACE SystemC::systemc)
add_library(eth_drv STATIC driver/src/eth_drv.c)
target_include_directories(eth_drv PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}/driver/include")
target_compile_features(eth_drv PUBLIC c_std_99)
foreach(_target eth_core eth_mac_tlm eth_drv)
    set_target_properties(${_target} PROPERTIES CXX_EXTENSIONS OFF C_EXTENSIONS OFF)
    if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
        target_compile_options(${_target} PRIVATE -Wall -Wextra -Wno-unused-parameter)
    endif()
endforeach()
# Test-only bus/memory/UART helpers are deliberately outside the public includes.
if(ICSL_ETH_VP_TESTS OR FX1_ETH_BUILD_EXAMPLES OR ICSL_ETH_VP_CDC)
    add_library(eth_vp_support INTERFACE)
    target_include_directories(eth_vp_support INTERFACE "${CMAKE_CURRENT_SOURCE_DIR}/tests/support/include")
    target_link_libraries(eth_vp_support INTERFACE SystemC::systemc)
endif()
if(ICSL_ETH_VP_TESTS)
    enable_testing()
endif()
if(FX1_ETH_BUILD_EXAMPLES OR ICSL_ETH_VP_TESTS)
    add_subdirectory(examples/eth_soc)
endif()
if(ICSL_ETH_VP_CDC)
    add_subdirectory(examples/cdc_vp)
endif()
if(ICSL_ETH_VP_TESTS)
    add_subdirectory(tests)
endif()
