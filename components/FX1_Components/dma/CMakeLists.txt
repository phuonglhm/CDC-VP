cmake_minimum_required(VERSION 3.16)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
  project(fx1_dma LANGUAGES CXX)
  include(CTest)
  set(_dma_tests_default ${BUILD_TESTING})
elseif(DEFINED CDC_BUILD_TESTS)
  set(_dma_tests_default ${CDC_BUILD_TESTS})
else()
  set(_dma_tests_default OFF)
endif()

if(NOT TARGET SystemC::systemc)
  if(SYSTEMC_HOME)
    list(PREPEND CMAKE_PREFIX_PATH "${SYSTEMC_HOME}")
  elseif(DEFINED ENV{SYSTEMC_HOME})
    list(PREPEND CMAKE_PREFIX_PATH "$ENV{SYSTEMC_HOME}")
  endif()
  find_package(SystemCLanguage CONFIG QUIET)
  if(NOT TARGET SystemC::systemc)
    find_path(SYSTEMC_INCLUDE_DIR NAMES systemc HINTS "${SYSTEMC_HOME}/include" "$ENV{SYSTEMC_HOME}/include")
    find_library(SYSTEMC_LIBRARY NAMES systemc HINTS "${SYSTEMC_HOME}/lib" "${SYSTEMC_HOME}/lib64" "$ENV{SYSTEMC_HOME}/lib")
    if(NOT SYSTEMC_INCLUDE_DIR OR NOT SYSTEMC_LIBRARY)
      message(FATAL_ERROR "SystemC not found. Set SYSTEMC_HOME or SYSTEMC_INCLUDE_DIR and SYSTEMC_LIBRARY.")
    endif()
    find_package(Threads REQUIRED)
    add_library(SystemC::systemc UNKNOWN IMPORTED)
    set_target_properties(SystemC::systemc PROPERTIES IMPORTED_LOCATION "${SYSTEMC_LIBRARY}"
      INTERFACE_INCLUDE_DIRECTORIES "${SYSTEMC_INCLUDE_DIR}"
      INTERFACE_LINK_LIBRARIES Threads::Threads)
  endif()
endif()

add_library(fx1_dma src/dma.cpp)
add_library(cdc::components::fx1_dma ALIAS fx1_dma)
target_compile_features(fx1_dma PUBLIC cxx_std_17)
target_include_directories(fx1_dma PUBLIC
  $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include> $<INSTALL_INTERFACE:include>)
target_link_libraries(fx1_dma PUBLIC SystemC::systemc)
if(MSVC)
  target_compile_options(fx1_dma PRIVATE /W4)
else()
  target_compile_options(fx1_dma PRIVATE -Wall -Wextra -Wpedantic)
endif()

option(FX1_DMA_BUILD_TESTS "Build DMA_Config regressions" ${_dma_tests_default})
# On by default when the FX1 bus is already part of the build (CDC-VP with
# CDC_BUILD_FX1_SOC); standalone it pulls the sibling bus in on request.
if(TARGET bus_system)
  set(_dma_bus_tests_default ON)
else()
  set(_dma_bus_tests_default OFF)
endif()
option(FX1_DMA_BUILD_BUS_TESTS "Test DMA through the sibling FX1 bus component" ${_dma_bus_tests_default})
if(FX1_DMA_BUILD_TESTS)
  enable_testing()
  add_executable(test_fx1_dma tests/test_dma.cpp)
  target_link_libraries(test_fx1_dma PRIVATE fx1_dma)
  foreach(case registers memory peripheral chain pending arbitration errors reset timing idle)
    add_test(NAME fx1_dma_${case} COMMAND test_fx1_dma ${case})
    set_tests_properties(fx1_dma_${case} PROPERTIES TIMEOUT 30 LABELS "fx1;fx1_unit")
  endforeach()
  # C firmware header vs the model's registers.h (static_asserts at compile time).
  add_executable(test_fx1_dma_regs_header tests/test_regs_header.cpp)
  target_include_directories(test_fx1_dma_regs_header PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/include)
  target_compile_features(test_fx1_dma_regs_header PRIVATE cxx_std_17)
  add_test(NAME fx1_dma_regs_header COMMAND test_fx1_dma_regs_header)
  set_tests_properties(fx1_dma_regs_header PROPERTIES TIMEOUT 10 LABELS "fx1;fx1_unit")
  if(FX1_DMA_BUILD_BUS_TESTS)
    if(NOT TARGET bus_system)
      option(BUS_SYSTEM_BUILD_TESTS "Build standalone bus regression" OFF)
      add_subdirectory("${CMAKE_CURRENT_SOURCE_DIR}/../bus" "${CMAKE_CURRENT_BINARY_DIR}/fx1-bus")
    endif()
    add_executable(test_fx1_dma_bus tests/test_bus.cpp)
    target_include_directories(test_fx1_dma_bus PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/../fx1_map/include)
    target_link_libraries(test_fx1_dma_bus PRIVATE fx1_dma bus_system)
    add_test(NAME fx1_dma_bus COMMAND test_fx1_dma_bus)
    set_tests_properties(fx1_dma_bus PROPERTIES TIMEOUT 30 LABELS "fx1;fx1_unit")
  endif()
endif()

if(CMAKE_PROJECT_NAME STREQUAL "cdc-vp")
  install(TARGETS fx1_dma EXPORT cdc-components-targets ARCHIVE DESTINATION lib
    LIBRARY DESTINATION lib RUNTIME DESTINATION bin)
else()
  install(TARGETS fx1_dma EXPORT fx1-dma-targets ARCHIVE DESTINATION lib
    LIBRARY DESTINATION lib RUNTIME DESTINATION bin)
  install(EXPORT fx1-dma-targets NAMESPACE cdc::components:: DESTINATION lib/cmake/fx1-dma)
  install(FILES cmake/fx1-dma-config.cmake DESTINATION lib/cmake/fx1-dma)
endif()
install(DIRECTORY include/ DESTINATION include)
