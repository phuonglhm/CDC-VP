# 4K (HAS maximum) end to end over TLM: the input is generated by the tool
# (tools/gen_4k_vectors.py pattern), the expected NV12 is a committed SHA-256
# and the statistics are checked through every readout mux (DEC-37).
file(GLOB _fx1_isp_4k_cases LIST_DIRECTORIES true ${CMAKE_CURRENT_SOURCE_DIR}/data/4k/*)
foreach(_dir IN LISTS _fx1_isp_4k_cases)
    if(IS_DIRECTORY ${_dir})
        get_filename_component(_case ${_dir} NAME)
        file(STRINGS ${_dir}/meta.txt _meta)
        separate_arguments(_meta UNIX_COMMAND "${_meta}")
        list(GET _meta 0 _w)
        list(GET _meta 1 _h)
        list(GET _meta 2 _frames)
        file(STRINGS ${_dir}/expected_nv12.sha256 _sha)
        separate_arguments(_sha UNIX_COMMAND "${_sha}")
        list(GET _sha 0 _sha)
        add_test(NAME fx1_isp.integration.e2e_4k.${_case}
            COMMAND fx1_isp_run_raw --synthetic --width ${_w} --height ${_h} --frames ${_frames}
                    --profile ${_dir}/profile.csrw --stats-script ${_dir}/expected_stats.txt --check-stats
                    --expect-sha256 ${_sha} --out ${CMAKE_CURRENT_BINARY_DIR}/e2e_4k_${_case})
    endif()
endforeach()

# The CSR artefacts in the tree must match the pinned spreadsheet. The
# spreadsheet is confidential and not in the repository, so this check runs
# only when its path is provided: -DFX1_ISP_CSR_XLSX=/path/FTEL_IP_ISP_CSR_v1.0.xlsx
set(FX1_ISP_CSR_XLSX "" CACHE FILEPATH "Path to FTEL_IP_ISP_CSR_v1.0.xlsx (optional)")
if(FX1_ISP_CSR_XLSX)
    # Prefer the `python3` on PATH over the newest versioned interpreter: the
    # tools need openpyxl, which is installed per interpreter.
    set(Python3_FIND_UNVERSIONED_NAMES FIRST)
    find_package(Python3 COMPONENTS Interpreter REQUIRED)
    execute_process(COMMAND ${Python3_EXECUTABLE} -c "import openpyxl"
                    RESULT_VARIABLE _fx1_isp_openpyxl OUTPUT_QUIET ERROR_QUIET)
    if(NOT _fx1_isp_openpyxl EQUAL 0)
        message(FATAL_ERROR "FX1_ISP_CSR_XLSX is set but ${Python3_EXECUTABLE} has no openpyxl. "
                            "Install it or pass -DPython3_EXECUTABLE=<python with openpyxl>.")
    endif()
    add_test(NAME fx1_isp.csr.generated_up_to_date
        COMMAND Python3::Interpreter ${PROJECT_SOURCE_DIR}/tools/gen_csr.py
                --xlsx ${FX1_ISP_CSR_XLSX} --out ${PROJECT_SOURCE_DIR} --check)
    add_test(NAME fx1_isp.csr.header_crosscheck
        COMMAND Python3::Interpreter ${PROJECT_SOURCE_DIR}/tools/check_csr_header.py
                --xlsx ${FX1_ISP_CSR_XLSX} --header ${PROJECT_SOURCE_DIR}/include/fx1_isp/fx1_isp_csr.h)
endif()

# when a python3 with numpy is found; the build itself never needs Python.

find_package(Python3 COMPONENTS Interpreter QUIET)
if(Python3_Interpreter_FOUND)
    # Per-register traceability report up to date, with its coverage gate.
    add_test(NAME fx1_isp.reference.traceability_up_to_date
        COMMAND Python3::Interpreter ${PROJECT_SOURCE_DIR}/tools/gen_traceability.py --check)
    execute_process(COMMAND ${Python3_EXECUTABLE} -c "import numpy"
                    RESULT_VARIABLE _fx1_isp_numpy OUTPUT_QUIET ERROR_QUIET)
    if(_fx1_isp_numpy EQUAL 0)
        add_test(NAME fx1_isp.reference.vectors_up_to_date
            COMMAND Python3::Interpreter ${PROJECT_SOURCE_DIR}/tools/gen_vectors.py --check)
        add_test(NAME fx1_isp.reference.block_vectors_up_to_date
            COMMAND Python3::Interpreter ${PROJECT_SOURCE_DIR}/tools/gen_block_vectors.py --check)
        add_test(NAME fx1_isp.reference.sequence_up_to_date
            COMMAND Python3::Interpreter ${PROJECT_SOURCE_DIR}/tools/gen_sequence.py --check)
        add_test(NAME fx1_isp.reference.4k_vectors_up_to_date
            COMMAND Python3::Interpreter ${PROJECT_SOURCE_DIR}/tools/gen_4k_vectors.py --check)
        set_tests_properties(fx1_isp.reference.4k_vectors_up_to_date PROPERTIES TIMEOUT 600)
        # Real-RAW regression (G-E2E/G-CONT, DEC-34..36): every inventory frame x
        # every preset plus the 28-frame sequences, bit-exact against the
        # reference and identical to the committed acceptance baseline. The dataset is not in
        # the repository: enable with -DFX1_ISP_RAW_INVENTORY=<ISP_RAW_INVENTORY.json>.
        set(FX1_ISP_RAW_INVENTORY "" CACHE FILEPATH "RAW inventory JSON for the real-RAW regression (optional)")
        set(FX1_ISP_RAW_OUT "${CMAKE_BINARY_DIR}/raw_regression" CACHE PATH "Artefact directory of the real-RAW regression")
        if(FX1_ISP_RAW_INVENTORY)
            add_test(NAME fx1_isp.regression.raw
                COMMAND Python3::Interpreter ${PROJECT_SOURCE_DIR}/tools/run_regression.py
                        --inventory ${FX1_ISP_RAW_INVENTORY} --run-raw $<TARGET_FILE:fx1_isp_run_raw>
                        --out ${FX1_ISP_RAW_OUT} --jobs 4
                        --baseline ${PROJECT_SOURCE_DIR}/docs/evidence/raw_regression_baseline.json)
            set_tests_properties(fx1_isp.regression.raw PROPERTIES TIMEOUT 3600)
        endif()
    else()
        message(STATUS "fx1_isp: numpy not found, reference vector checks disabled")
    endif()
endif()
