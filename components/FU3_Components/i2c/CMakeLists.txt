cmake_minimum_required(VERSION 3.16)
project(fu3_i2c VERSION 0.1.0 LANGUAGES CXX)
include(CTest)
if(NOT TARGET SystemC::systemc)
  if(SYSTEMC_HOME)
    list(PREPEND CMAKE_PREFIX_PATH "${SYSTEMC_HOME}")
  endif()
  find_package(SystemCLanguage CONFIG REQUIRED)
endif()
add_library(fu3_i2c src/i2c.cpp)
add_library(fu3::i2c ALIAS fu3_i2c)
set_target_properties(fu3_i2c PROPERTIES EXPORT_NAME i2c POSITION_INDEPENDENT_CODE ON WINDOWS_EXPORT_ALL_SYMBOLS ON)
target_compile_features(fu3_i2c PUBLIC cxx_std_17)
target_include_directories(fu3_i2c PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include> $<INSTALL_INTERFACE:include>)
target_link_libraries(fu3_i2c PUBLIC SystemC::systemc)
if(NOT MSVC)
  target_compile_options(fu3_i2c PRIVATE -Wall -Wextra -Wpedantic)
endif()
if(BUILD_TESTING)
  set(FU3_BUS_BUILD_TESTS OFF CACHE BOOL "" FORCE)
  if(NOT TARGET fu3::bus_system)
    add_subdirectory(../bus bus)
  endif()
  foreach(kind behavior integration)
    add_executable(i2c_${kind}_tests tests/${kind}.cpp)
    target_link_libraries(i2c_${kind}_tests PRIVATE fu3::i2c fu3::bus_system)
    add_test(NAME i2c_${kind} COMMAND i2c_${kind}_tests)
    set_tests_properties(i2c_${kind} PROPERTIES TIMEOUT 30)
  endforeach()
endif()
install(TARGETS fu3_i2c EXPORT fu3-i2c-targets ARCHIVE DESTINATION lib LIBRARY DESTINATION lib RUNTIME DESTINATION bin)
install(EXPORT fu3-i2c-targets NAMESPACE fu3:: DESTINATION lib/cmake/fu3-i2c)
install(FILES cmake/fu3-i2c-config.cmake DESTINATION lib/cmake/fu3-i2c)
install(DIRECTORY include/ DESTINATION include)
