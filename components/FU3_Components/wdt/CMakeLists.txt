cmake_minimum_required(VERSION 3.16)
project(fu3_wdt VERSION 0.1.0 LANGUAGES CXX)
include(CTest)
if(NOT TARGET SystemC::systemc)
  if(SYSTEMC_HOME)
    list(PREPEND CMAKE_PREFIX_PATH "${SYSTEMC_HOME}")
  endif()
  find_package(SystemCLanguage CONFIG REQUIRED)
endif()
add_library(fu3_wdt src/wdt.cpp)
add_library(fu3::wdt ALIAS fu3_wdt)
set_target_properties(fu3_wdt PROPERTIES EXPORT_NAME wdt POSITION_INDEPENDENT_CODE ON WINDOWS_EXPORT_ALL_SYMBOLS ON)
target_compile_features(fu3_wdt PUBLIC cxx_std_17)
target_include_directories(fu3_wdt PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include> $<INSTALL_INTERFACE:include>)
target_link_libraries(fu3_wdt PUBLIC SystemC::systemc)
if(NOT MSVC)
  target_compile_options(fu3_wdt PRIVATE -Wall -Wextra -Wpedantic)
endif()
if(BUILD_TESTING)
  if(NOT TARGET fu3::bus_system)
    # Own copy of the bus: skip its tests in this scope only, never in the cache.
    set(FU3_BUS_BUILD_TESTS OFF)
    add_subdirectory(../bus bus)
  endif()
  foreach(kind behavior integration)
    add_executable(wdt_${kind}_tests tests/${kind}.cpp)
    target_link_libraries(wdt_${kind}_tests PRIVATE fu3::wdt fu3::bus_system)
    add_test(NAME wdt_${kind} COMMAND wdt_${kind}_tests)
    set_tests_properties(wdt_${kind} PROPERTIES TIMEOUT 30)
  endforeach()
endif()
install(TARGETS fu3_wdt EXPORT fu3-wdt-targets ARCHIVE DESTINATION lib LIBRARY DESTINATION lib RUNTIME DESTINATION bin)
install(EXPORT fu3-wdt-targets NAMESPACE fu3:: DESTINATION lib/cmake/fu3-wdt)
install(FILES cmake/fu3-wdt-config.cmake DESTINATION lib/cmake/fu3-wdt)
install(DIRECTORY include/ DESTINATION include)
