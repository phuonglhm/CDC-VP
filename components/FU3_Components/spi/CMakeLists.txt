cmake_minimum_required(VERSION 3.16)
project(fu3_spi VERSION 0.1.0 LANGUAGES CXX)
include(CTest)
if(NOT TARGET SystemC::systemc)
  if(SYSTEMC_HOME)
    list(PREPEND CMAKE_PREFIX_PATH "${SYSTEMC_HOME}")
  endif()
  find_package(SystemCLanguage CONFIG REQUIRED)
endif()
add_library(fu3_spi src/spi.cpp)
add_library(fu3::spi ALIAS fu3_spi)
set_target_properties(fu3_spi PROPERTIES EXPORT_NAME spi POSITION_INDEPENDENT_CODE ON WINDOWS_EXPORT_ALL_SYMBOLS ON)
target_compile_features(fu3_spi PUBLIC cxx_std_17)
target_include_directories(fu3_spi PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include> $<INSTALL_INTERFACE:include>)
target_link_libraries(fu3_spi PUBLIC SystemC::systemc)
if(NOT MSVC)
  target_compile_options(fu3_spi PRIVATE -Wall -Wextra -Wpedantic)
endif()
if(BUILD_TESTING)
  set(FU3_BUS_BUILD_TESTS OFF CACHE BOOL "" FORCE)
  if(NOT TARGET fu3::bus_system)
    add_subdirectory(../bus bus)
  endif()
  foreach(kind behavior integration)
    add_executable(spi_${kind}_tests tests/${kind}.cpp)
    target_link_libraries(spi_${kind}_tests PRIVATE fu3::spi fu3::bus_system)
    add_test(NAME spi_${kind} COMMAND spi_${kind}_tests)
    set_tests_properties(spi_${kind} PROPERTIES TIMEOUT 30)
  endforeach()
  foreach(depth 32 64)
    add_test(NAME spi_behavior_depth_${depth} COMMAND spi_behavior_tests ${depth})
    set_tests_properties(spi_behavior_depth_${depth} PROPERTIES TIMEOUT 30)
  endforeach()
endif()
install(TARGETS fu3_spi EXPORT fu3-spi-targets ARCHIVE DESTINATION lib LIBRARY DESTINATION lib RUNTIME DESTINATION bin)
install(EXPORT fu3-spi-targets NAMESPACE fu3:: DESTINATION lib/cmake/fu3-spi)
install(FILES cmake/fu3-spi-config.cmake DESTINATION lib/cmake/fu3-spi)
install(DIRECTORY include/ DESTINATION include)
