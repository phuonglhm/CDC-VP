cmake_minimum_required(VERSION 3.16)
project(spi_consumer LANGUAGES CXX)
find_package(fu3-spi CONFIG REQUIRED)
add_executable(spi_consumer ../behavior.cpp)
target_link_libraries(spi_consumer PRIVATE fu3::spi)
enable_testing()
add_test(NAME spi_installed_consumer COMMAND spi_consumer)
set_tests_properties(spi_installed_consumer PROPERTIES TIMEOUT 30)
add_executable(spi_corner_consumer ../corners.cpp)
target_link_libraries(spi_corner_consumer PRIVATE fu3::spi)
foreach(scenario slave_formats fifo_flush event_priority abort_states dma_boundaries reset_phases master_delays)
  add_test(NAME spi_installed_${scenario} COMMAND spi_corner_consumer ${scenario} 16 1)
  set_tests_properties(spi_installed_${scenario} PROPERTIES TIMEOUT 30)
endforeach()
