cmake_minimum_required(VERSION 3.16)
project(fu3_scoped_installed_validation LANGUAGES CXX)
enable_testing()
find_package(fu3-bus CONFIG REQUIRED)
get_filename_component(FU3_ROOT "${CMAKE_CURRENT_LIST_DIR}/../../.." ABSOLUTE)
foreach(ip timer wdt uart spi i2c hyperram rng)
  find_package(fu3-${ip} CONFIG REQUIRED)
  add_executable(installed_${ip} "${FU3_ROOT}/${ip}/tests/behavior.cpp")
  target_link_libraries(installed_${ip} PRIVATE fu3::${ip} fu3::bus_system)
  add_test(NAME installed_${ip} COMMAND installed_${ip})
  set_tests_properties(installed_${ip} PROPERTIES TIMEOUT 30)
endforeach()
add_executable(installed_spi_corners "${FU3_ROOT}/spi/tests/corners.cpp")
target_link_libraries(installed_spi_corners PRIVATE fu3::spi)
foreach(scenario slave_formats fifo_flush event_priority abort_states dma_boundaries reset_phases master_delays)
  add_test(NAME installed_spi_${scenario} COMMAND installed_spi_corners ${scenario} 16 1)
  set_tests_properties(installed_spi_${scenario} PROPERTIES TIMEOUT 30)
endforeach()
