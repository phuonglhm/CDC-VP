# Production-IP dependencies belong to tests, not the exported bus library.
function(fu3_bus_test_ip component binary)
  set(BUILD_TESTING OFF)
  set(FU3_BUS_BUILD_TESTS OFF)
  add_subdirectory("${CMAKE_CURRENT_SOURCE_DIR}/../../../${component}" "${binary}")
endfunction()
if(NOT TARGET fu3_ap_dma)
  fu3_bus_test_ip("DMA0/AP_DMA" dma)
endif()
foreach(ip spi uart timer wdt plic)
  if(NOT TARGET fu3::${ip})
    fu3_bus_test_ip("${ip}" "${ip}")
  endif()
endforeach()
add_executable(fu3_bus_real_ip_tests real_ip_tests.cpp)
target_include_directories(fu3_bus_real_ip_tests PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/..")
target_link_libraries(fu3_bus_real_ip_tests PRIVATE fu3::bus_system fu3_ap_dma
  fu3::spi fu3::uart fu3::timer fu3::wdt fu3::plic)
foreach(scenario adapter dma plic reset)
  add_test(NAME fu3_bus_real_${scenario} COMMAND fu3_bus_real_ip_tests ${scenario})
  set_tests_properties(fu3_bus_real_${scenario} PROPERTIES TIMEOUT 30)
endforeach()
