cmake_minimum_required(VERSION 3.16)
project(fu3_scoped_vp_validation LANGUAGES CXX)
include(CTest)
get_filename_component(FU3_ROOT "${CMAKE_CURRENT_LIST_DIR}/../../.." ABSOLUTE)
set(FU3_BUS_BUILD_TESTS ON CACHE BOOL "" FORCE)
set(FU3_BUS_BUILD_IP_TESTS OFF CACHE BOOL "" FORCE)
add_subdirectory("${FU3_ROOT}/bus" bus)
foreach(ip timer wdt uart spi i2c hyperram rng)
  add_subdirectory("${FU3_ROOT}/${ip}" "${ip}")
endforeach()
# Other owners' models are read-only peers. All generated output stays here.
function(add_peer component binary)
  set(BUILD_TESTING OFF)
  add_subdirectory("${FU3_ROOT}/${component}" "${binary}")
endfunction()
add_peer("DMA0/AP_DMA" dma_peer)
add_peer(plic plic_peer)
add_subdirectory("${FU3_ROOT}/bus/tests/ip_integration" real_ip)
option(FU3_BUS_BUILD_PULLED_PEER_TESTS "Test bus interoperability with current QSPI and GPIO peers" ON)
if(FU3_BUS_BUILD_PULLED_PEER_TESTS)
  function(add_pulled_peer component binary)
    set(BUILD_TESTING OFF)
    set(FU3_QSPI_BUILD_TESTS OFF)
    set(FU3_GPIO_BUILD_TESTS OFF)
    set(FU3_IO_PAD_BUILD_TESTS OFF)
    add_subdirectory("${FU3_ROOT}/${component}" "${binary}")
  endfunction()
  add_pulled_peer(QSPI qspi_peer)
  add_pulled_peer(GPIO gpio_peer)
  add_subdirectory("${FU3_ROOT}/bus/tests/pulled_peers" pulled_peers)
endif()
