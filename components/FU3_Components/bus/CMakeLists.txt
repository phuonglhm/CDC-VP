cmake_minimum_required(VERSION 3.16)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
  project(fu3_bus_system VERSION 0.1.0 LANGUAGES CXX)
  include(CTest)
endif()
if(NOT TARGET SystemC::systemc)
  if(SYSTEMC_HOME)
    list(PREPEND CMAKE_PREFIX_PATH "${SYSTEMC_HOME}")
  elseif(DEFINED ENV{SYSTEMC_HOME})
    list(PREPEND CMAKE_PREFIX_PATH "$ENV{SYSTEMC_HOME}")
  endif()
  find_package(SystemCLanguage CONFIG REQUIRED)
endif()
add_library(fu3_bus_system src/config.cpp src/router.cpp src/apb_bridge.cpp src/bus_system.cpp src/initiator_port.cpp)
add_library(fu3::bus_system ALIAS fu3_bus_system)
set_target_properties(fu3_bus_system PROPERTIES EXPORT_NAME bus_system
  POSITION_INDEPENDENT_CODE ON WINDOWS_EXPORT_ALL_SYMBOLS ON)
target_compile_features(fu3_bus_system PUBLIC cxx_std_17)
target_include_directories(fu3_bus_system PUBLIC
  $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include>
  $<INSTALL_INTERFACE:include>)
target_link_libraries(fu3_bus_system PUBLIC SystemC::systemc)
if(MSVC)
  target_compile_options(fu3_bus_system PRIVATE /W4)
else()
  target_compile_options(fu3_bus_system PRIVATE -Wall -Wextra -Wpedantic)
endif()
option(FU3_BUS_BUILD_TESTS "Build FU3 behavioral regression" ${BUILD_TESTING})
if(FU3_BUS_BUILD_TESTS)
  enable_testing()
  foreach(kind config behavior arbitration contract safety)
    add_executable(fu3_bus_${kind}_tests tests/bus_${kind}_tests.cpp)
    target_link_libraries(fu3_bus_${kind}_tests PRIVATE fu3::bus_system)
    add_test(NAME fu3_bus_${kind} COMMAND fu3_bus_${kind}_tests)
    set_tests_properties(fu3_bus_${kind} PROPERTIES TIMEOUT 30)
  endforeach()
  add_test(NAME fu3_bus_zero_latency_fairness COMMAND fu3_bus_arbitration_tests --zero-latency)
  set_tests_properties(fu3_bus_zero_latency_fairness PROPERTIES TIMEOUT 30)
  option(FU3_BUS_BUILD_IP_TESTS "Test the bus with production FU3 DMA and peripheral models" OFF)
  if(FU3_BUS_BUILD_IP_TESTS)
    add_subdirectory(tests/ip_integration)
  endif()
endif()
install(TARGETS fu3_bus_system EXPORT fu3-bus-targets
  ARCHIVE DESTINATION lib LIBRARY DESTINATION lib RUNTIME DESTINATION bin)
install(EXPORT fu3-bus-targets NAMESPACE fu3:: DESTINATION lib/cmake/fu3-bus)
install(FILES cmake/fu3-bus-config.cmake DESTINATION lib/cmake/fu3-bus)
install(DIRECTORY include/ DESTINATION include)
