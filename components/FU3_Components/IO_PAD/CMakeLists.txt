cmake_minimum_required(VERSION 3.16)
if(TARGET fu3::io_pad)
  return()
endif()
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
  project(fu3_io_pad LANGUAGES CXX)
  include(CTest)
endif()
if(NOT TARGET SystemC::systemc)
  set(SYSTEMC_HOME "/opt/systemc-2.3.4" CACHE PATH "SystemC installation")
  find_path(SYSTEMC_INCLUDE_DIR systemc HINTS "${SYSTEMC_HOME}/include")
  find_library(SYSTEMC_LIBRARY systemc HINTS "${SYSTEMC_HOME}/lib" "${SYSTEMC_HOME}/lib64")
  if(NOT SYSTEMC_INCLUDE_DIR OR NOT SYSTEMC_LIBRARY)
    message(FATAL_ERROR "Set SYSTEMC_HOME to an installed SystemC prefix")
  endif()
  if(NOT TARGET Threads::Threads)
    find_package(Threads REQUIRED)
    # Keep imported dependencies visible when GPIO and IO_PAD are siblings.
    set_property(TARGET Threads::Threads PROPERTY IMPORTED_GLOBAL TRUE)
  endif()
  add_library(SystemC::systemc UNKNOWN IMPORTED GLOBAL)
  set_target_properties(SystemC::systemc PROPERTIES IMPORTED_LOCATION "${SYSTEMC_LIBRARY}"
    INTERFACE_INCLUDE_DIRECTORIES "${SYSTEMC_INCLUDE_DIR}"
    INTERFACE_LINK_LIBRARIES "Threads::Threads;${CMAKE_DL_LIBS}")
endif()

option(FU3_GPIO_COVERAGE "Enable GCC coverage for GPIO, IO_PAD and their subsystem" OFF)
if(FU3_GPIO_COVERAGE AND NOT CMAKE_CXX_COMPILER_ID STREQUAL "GNU")
  message(FATAL_ERROR "FU3_GPIO_COVERAGE requires GCC/gcov")
endif()

add_library(fu3_io_pad src/io_pad.cpp)
add_library(fu3::io_pad ALIAS fu3_io_pad)
target_compile_features(fu3_io_pad PUBLIC cxx_std_17)
target_include_directories(fu3_io_pad PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}/include")
target_link_libraries(fu3_io_pad PUBLIC SystemC::systemc)
if(MSVC)
  target_compile_options(fu3_io_pad PRIVATE /W4)
else()
  target_compile_options(fu3_io_pad PRIVATE -Wall -Wextra -Wpedantic)
endif()
if(FU3_GPIO_COVERAGE)
  target_compile_options(fu3_io_pad PRIVATE --coverage -O0 -g)
  target_link_options(fu3_io_pad PUBLIC --coverage)
endif()

if(DEFINED CDC_BUILD_TESTS)
  set(_fu3_io_pad_tests_default "${CDC_BUILD_TESTS}")
else()
  set(_fu3_io_pad_tests_default "${BUILD_TESTING}")
endif()
if(DEFINED FU3_GPIO_BUILD_TESTS)
  set(_fu3_io_pad_tests_default ${FU3_GPIO_BUILD_TESTS})
endif()
option(FU3_IO_PAD_BUILD_TESTS "Build standalone IO_PAD tests" ${_fu3_io_pad_tests_default})
if(FU3_IO_PAD_BUILD_TESTS)
  enable_testing()
  add_executable(test_fu3_io_pad tests/test_io_pad.cpp)
  target_link_libraries(test_fu3_io_pad PRIVATE fu3::io_pad)
  foreach(scenario truth random metadata isolation delta invalid_partition)
    add_test(NAME fu3_io_pad_${scenario} COMMAND test_fu3_io_pad pad_${scenario})
    set_tests_properties(fu3_io_pad_${scenario} PROPERTIES TIMEOUT 30 LABELS "fu3;io_pad;io;spec")
  endforeach()
endif()
