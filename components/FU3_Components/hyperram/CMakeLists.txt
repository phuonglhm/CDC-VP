cmake_minimum_required(VERSION 3.16)
project(fu3_hyperram VERSION 0.1.0 LANGUAGES CXX)
include(CTest)
if(NOT TARGET fu3::bus_system)
  add_subdirectory(../bus bus)
endif()
add_library(fu3_hyperram src/controller.cpp)
add_library(fu3::hyperram ALIAS fu3_hyperram)
set_target_properties(fu3_hyperram PROPERTIES EXPORT_NAME hyperram POSITION_INDEPENDENT_CODE ON WINDOWS_EXPORT_ALL_SYMBOLS ON)
target_compile_features(fu3_hyperram PUBLIC cxx_std_17)
target_include_directories(fu3_hyperram PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include> $<INSTALL_INTERFACE:include>)
target_link_libraries(fu3_hyperram PUBLIC fu3::bus_system)
if(NOT MSVC)
  target_compile_options(fu3_hyperram PRIVATE -Wall -Wextra -Wpedantic)
endif()
if(BUILD_TESTING)
  add_executable(hyperram_tests tests/behavior.cpp)
  target_link_libraries(hyperram_tests PRIVATE fu3::hyperram)
  add_test(NAME hyperram_behavior COMMAND hyperram_tests direct)
  add_test(NAME hyperram_bus_integration COMMAND hyperram_tests bus)
  set_tests_properties(hyperram_behavior hyperram_bus_integration PROPERTIES TIMEOUT 30)
endif()
install(TARGETS fu3_hyperram EXPORT fu3-hyperram-targets ARCHIVE DESTINATION lib LIBRARY DESTINATION lib RUNTIME DESTINATION bin)
install(EXPORT fu3-hyperram-targets NAMESPACE fu3:: DESTINATION lib/cmake/fu3-hyperram)
install(FILES cmake/fu3-hyperram-config.cmake DESTINATION lib/cmake/fu3-hyperram)
install(DIRECTORY include/ DESTINATION include)
