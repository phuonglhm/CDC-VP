cmake_minimum_required(VERSION 3.16)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
  project(fu3_adc LANGUAGES CXX)
  include(CTest)
endif()
if(NOT TARGET SystemC::systemc)
  set(SYSTEMC_HOME "/opt/systemc-2.3.4" CACHE PATH "SystemC installation prefix")
  find_path(SYSTEMC_INCLUDE_DIR systemc HINTS "${SYSTEMC_HOME}/include")
  find_library(SYSTEMC_LIBRARY systemc HINTS "${SYSTEMC_HOME}/lib" "${SYSTEMC_HOME}/lib64")
  if(NOT SYSTEMC_INCLUDE_DIR OR NOT SYSTEMC_LIBRARY)
    message(FATAL_ERROR "Set SYSTEMC_HOME to an installed SystemC prefix")
  endif()
  find_package(Threads REQUIRED)
  add_library(SystemC::systemc UNKNOWN IMPORTED)
  set_target_properties(SystemC::systemc PROPERTIES
    IMPORTED_LOCATION "${SYSTEMC_LIBRARY}"
    INTERFACE_INCLUDE_DIRECTORIES "${SYSTEMC_INCLUDE_DIR}"
    INTERFACE_LINK_LIBRARIES "Threads::Threads;${CMAKE_DL_LIBS}")
endif()
add_library(fu3_adc src/adc.cpp)
add_library(fu3::adc ALIAS fu3_adc)
target_compile_features(fu3_adc PUBLIC cxx_std_17)
target_include_directories(fu3_adc PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include>)
target_link_libraries(fu3_adc PUBLIC SystemC::systemc)
option(FU3_ADC_BUILD_EXAMPLE "Build the minimal polling ADC VP" ON)
if(FU3_ADC_BUILD_EXAMPLE)
  add_executable(fu3_adc_vp examples/adc_vp.cpp)
  target_link_libraries(fu3_adc_vp PRIVATE fu3::adc)
endif()
if(MSVC)
  target_compile_options(fu3_adc PRIVATE /W4)
else()
  target_compile_options(fu3_adc PRIVATE -Wall -Wextra -Wpedantic)
endif()
option(FU3_ADC_COVERAGE "Instrument ADC source with GCC coverage" OFF)
if(FU3_ADC_COVERAGE)
  if(NOT CMAKE_CXX_COMPILER_ID STREQUAL "GNU")
    message(FATAL_ERROR "FU3_ADC_COVERAGE requires GCC")
  endif()
  target_compile_options(fu3_adc PRIVATE --coverage -O0 -g)
  target_link_options(fu3_adc PUBLIC --coverage)
endif()
set(_adc_tests ${BUILD_TESTING})
if(DEFINED CDC_BUILD_TESTS)
  set(_adc_tests ${CDC_BUILD_TESTS})
endif()
option(FU3_ADC_BUILD_TESTS "Build ADC verification" ${_adc_tests})
if(FU3_ADC_BUILD_TESTS)
  enable_testing()
  if(FU3_ADC_BUILD_EXAMPLE)
    add_test(NAME fu3_adc_example COMMAND fu3_adc_vp)
    set_tests_properties(fu3_adc_example PROPERTIES TIMEOUT 30 LABELS "fu3;adc;example")
  endif()
  add_executable(test_fu3_adc tests/test_adc.cpp)
  target_link_libraries(test_fu3_adc PRIVATE fu3::adc)
  foreach(scenario registers channels counts timing_min timing_programmed timing_max
      averaging eoc_wait eoc_policy lifecycle snapshot reset protocol debug randomized clock_pause
      reset_phases disable_phases reset_transport)
    add_test(NAME fu3_adc_${scenario} COMMAND test_fu3_adc ${scenario})
    set_tests_properties(fu3_adc_${scenario} PROPERTIES TIMEOUT 30 LABELS "fu3;adc")
  endforeach()
  option(FU3_ADC_BUS_TESTS "Test the actual FU3 SYSBUS_1/PERIBUS_1 path" ON)
  if(FU3_ADC_BUS_TESTS)
    if(NOT TARGET fu3_bus_system)
      set(FU3_BUS_BUILD_TESTS OFF)
      add_subdirectory(../bus "${CMAKE_CURRENT_BINARY_DIR}/fu3-bus")
    endif()
    add_executable(test_fu3_adc_bus tests/test_adc.cpp)
    target_compile_definitions(test_fu3_adc_bus PRIVATE FU3_ADC_BUS_TEST=1)
    target_link_libraries(test_fu3_adc_bus PRIVATE fu3::adc fu3::bus_system)
    foreach(scenario registers channels counts averaging lifecycle snapshot bus)
      add_test(NAME fu3_adc_bus_${scenario} COMMAND test_fu3_adc_bus ${scenario})
      set_tests_properties(fu3_adc_bus_${scenario} PROPERTIES TIMEOUT 30 LABELS "fu3;adc;integration")
    endforeach()
  endif()
endif()
