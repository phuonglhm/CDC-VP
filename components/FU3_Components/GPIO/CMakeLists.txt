cmake_minimum_required(VERSION 3.16)
if(TARGET fu3::gpio)
  return()
endif()
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
  project(fu3_gpio LANGUAGES CXX)
  include(CTest)
endif()
if(DEFINED CDC_BUILD_TESTS)
  set(_fu3_gpio_tests_default "${CDC_BUILD_TESTS}")
else()
  set(_fu3_gpio_tests_default "${BUILD_TESTING}")
endif()
option(FU3_GPIO_BUILD_TESTS "Build GPIO core and integration tests" ${_fu3_gpio_tests_default})
if(FU3_GPIO_BUILD_TESTS)
  enable_testing()
endif()
if(NOT TARGET fu3::io_pad)
  add_subdirectory(../IO_PAD "${CMAKE_CURRENT_BINARY_DIR}/io_pad")
endif()
add_library(fu3_gpio src/state.cpp src/gpio.cpp src/subsystem.cpp)
add_library(fu3::gpio ALIAS fu3_gpio)
# Legacy target name resolves to GPIO; no separate integration library exists.
add_library(fu3::gpio_io ALIAS fu3_gpio)
target_compile_features(fu3_gpio PUBLIC cxx_std_17)
target_include_directories(fu3_gpio PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}/include")
target_link_libraries(fu3_gpio PUBLIC fu3::io_pad)
if(MSVC)
  target_compile_options(fu3_gpio PRIVATE /W4)
else()
  target_compile_options(fu3_gpio PRIVATE -Wall -Wextra -Wpedantic)
endif()
if(FU3_GPIO_COVERAGE)
  target_compile_options(fu3_gpio PRIVATE --coverage -O0 -g)
  target_link_options(fu3_gpio PUBLIC --coverage)
endif()

if(FU3_GPIO_BUILD_TESTS)
  add_executable(test_fu3_gpio_spec tests/test_spec.cpp)
  target_link_libraries(test_fu3_gpio_spec PRIVATE fu3::gpio)
  foreach(scenario register_outputs input_stream irq_boundaries irq_histories wake_priority clock_reset address_decode address_writes)
    add_test(NAME fu3_gpio_spec_${scenario} COMMAND test_fu3_gpio_spec ${scenario})
    set_tests_properties(fu3_gpio_spec_${scenario} PROPERTIES TIMEOUT 30 LABELS "fu3;gpio;spec")
  endforeach()
  add_executable(test_fu3_gpio_integration tests/test_integration.cpp)
  target_link_libraries(test_fu3_gpio_integration PRIVATE fu3::gpio)
  foreach(scenario registers sampling interrupts wake protocol reset concurrency reset_pulse iosel integration wake_iosel random)
    add_test(NAME fu3_gpio_${scenario} COMMAND test_fu3_gpio_integration ${scenario})
    set_tests_properties(fu3_gpio_${scenario} PROPERTIES TIMEOUT 30 LABELS "fu3;gpio;io;integration")
  endforeach()
  add_executable(test_fu3_gpio_apb tests/test_apb.cpp)
  target_link_libraries(test_fu3_gpio_apb PRIVATE fu3::gpio)
  if(FU3_GPIO_COVERAGE)
    target_compile_options(test_fu3_gpio_apb PRIVATE --coverage -O0 -g)
  endif()
  foreach(scenario registers phases interrupts sampling_wake reset random)
    add_test(NAME fu3_gpio_apb_${scenario} COMMAND test_fu3_gpio_apb ${scenario})
    set_tests_properties(fu3_gpio_apb_${scenario} PROPERTIES TIMEOUT 30 LABELS "fu3;gpio;apb;spec")
  endforeach()
endif()
