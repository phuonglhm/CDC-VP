cmake_minimum_required(VERSION 3.16)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
  project(fu3_qspi LANGUAGES CXX)
  include(CTest)
endif()
if(NOT TARGET SystemC::systemc)
  set(SYSTEMC_HOME "/opt/systemc-2.3.4" CACHE PATH "SystemC installation")
  find_path(SYSTEMC_INCLUDE_DIR systemc HINTS "${SYSTEMC_HOME}/include")
  find_library(SYSTEMC_LIBRARY systemc HINTS "${SYSTEMC_HOME}/lib" "${SYSTEMC_HOME}/lib64")
  if(NOT SYSTEMC_INCLUDE_DIR OR NOT SYSTEMC_LIBRARY)
    message(FATAL_ERROR "Set SYSTEMC_HOME to an installed SystemC prefix")
  endif()
  find_package(Threads REQUIRED)
  add_library(SystemC::systemc UNKNOWN IMPORTED)
  set_target_properties(SystemC::systemc PROPERTIES IMPORTED_LOCATION "${SYSTEMC_LIBRARY}"
    INTERFACE_INCLUDE_DIRECTORIES "${SYSTEMC_INCLUDE_DIR}")
  set_property(TARGET SystemC::systemc PROPERTY INTERFACE_LINK_LIBRARIES "Threads::Threads;${CMAKE_DL_LIBS}")
endif()
add_library(fu3_qspi src/qspi_controller.cpp src/qspi_flash_model.cpp)
add_library(fu3::qspi ALIAS fu3_qspi)
target_compile_features(fu3_qspi PUBLIC cxx_std_17)
target_include_directories(fu3_qspi PUBLIC include)
target_link_libraries(fu3_qspi PUBLIC SystemC::systemc)
if(MSVC)
  target_compile_options(fu3_qspi PRIVATE /W4)
else()
  target_compile_options(fu3_qspi PRIVATE -Wall -Wextra -Wpedantic)
endif()
set(_qspi_tests ${BUILD_TESTING})
if(DEFINED CDC_BUILD_TESTS)
  set(_qspi_tests ${CDC_BUILD_TESTS})
endif()
option(FU3_QSPI_BUILD_TESTS "Build QSPI spec regression" ${_qspi_tests})
if(FU3_QSPI_BUILD_TESTS)
  enable_testing()
  add_executable(test_fu3_qspi tests/test_qspi.cpp)
  target_link_libraries(test_fu3_qspi PRIVATE fu3::qspi)
  if(NOT MSVC)
    target_compile_options(test_fu3_qspi PRIVATE -Wall -Wextra -Wpedantic)
  endif()
  foreach(scenario registers protocol triggers indirect fifo polling timeout dma xip phases errors reset clock concurrency random components threshold_words spec_timing)
    add_test(NAME fu3_qspi_${scenario} COMMAND test_fu3_qspi ${scenario})
    set_tests_properties(fu3_qspi_${scenario} PROPERTIES TIMEOUT 30 LABELS "fu3;qspi;spec")
  endforeach()
  add_executable(test_fu3_qspi_corners tests/test_corners.cpp)
  target_link_libraries(test_fu3_qspi_corners PRIVATE fu3::qspi)
  if(NOT MSVC)
    target_compile_options(test_fu3_qspi_corners PRIVATE -Wall -Wextra -Wpedantic)
  endif()
  foreach(scenario p0_triggers p0_dlr p0_fifo p0_irq_w1c p0_active_reset p0_bus_errors
                   p1_polling p1_dma p1_domains hds_cache hds_control hds_dma_clock hds_flags hds_timeout hds_abort
                   hds_frame_matrix hds_dividers hds_parameters)
    add_test(NAME fu3_qspi_${scenario} COMMAND test_fu3_qspi_corners ${scenario})
    if(scenario MATCHES "^hds_")
      set(priority hds)
    else()
      string(SUBSTRING "${scenario}" 0 2 priority)
    endif()
    set_tests_properties(fu3_qspi_${scenario} PROPERTIES TIMEOUT 30 LABELS "fu3;qspi;corners;${priority}")
  endforeach()
  foreach(count 1 2 3 4 8 16)
    add_test(NAME fu3_qspi_hds_cache_parameters_${count}
      COMMAND test_fu3_qspi_corners hds_cache_parameters_${count} 32 32 ${count})
    set_tests_properties(fu3_qspi_hds_cache_parameters_${count} PROPERTIES TIMEOUT 30 LABELS "fu3;qspi;hds;parameters")
  endforeach()
  foreach(depth 4 8 16 32)
    add_test(NAME fu3_qspi_p1_parameters_${depth}
      COMMAND test_fu3_qspi_corners p1_parameters_${depth} ${depth})
    set_tests_properties(fu3_qspi_p1_parameters_${depth} PROPERTIES TIMEOUT 30 LABELS "fu3;qspi;corners;p1;parameters")
  endforeach()
  foreach(depths 4x32 8x32)
    string(REPLACE "x" ";" pair "${depths}")
    list(GET pair 0 tx_depth)
    list(GET pair 1 rx_depth)
    add_test(NAME fu3_qspi_p1_parameters_${depths}
      COMMAND test_fu3_qspi_corners p1_parameters_${depths} ${tx_depth} ${rx_depth})
    set_tests_properties(fu3_qspi_p1_parameters_${depths} PROPERTIES TIMEOUT 30 LABELS "fu3;qspi;corners;p1;parameters")
  endforeach()
  add_executable(test_fu3_qspi_flash tests/test_flash.cpp)
  target_link_libraries(test_fu3_qspi_flash PRIVATE fu3::qspi)
  foreach(scenario flash_commands flash_program flash_erase flash_protocol flash_reset flash_image flash_sioo flash_integrity)
    add_test(NAME fu3_qspi_${scenario} COMMAND test_fu3_qspi_flash ${scenario})
    set_tests_properties(fu3_qspi_${scenario} PROPERTIES TIMEOUT 30 LABELS "fu3;qspi;flash")
  endforeach()
  set_property(TEST fu3_qspi_flash_integrity APPEND PROPERTY LABELS "p1;corners")
  option(FU3_QSPI_BUS_TESTS "Test integration with the FU3 bus" ON)
  if(FU3_QSPI_BUS_TESTS)
    if(NOT TARGET fu3_bus_system)
      set(FU3_BUS_BUILD_TESTS OFF)
      add_subdirectory(../bus "${CMAKE_CURRENT_BINARY_DIR}/fu3-bus")
    endif()
    add_executable(test_fu3_qspi_bus tests/test_bus.cpp)
    target_link_libraries(test_fu3_qspi_bus PRIVATE fu3::qspi fu3::bus_system)
    add_test(NAME fu3_qspi_bus COMMAND test_fu3_qspi_bus)
    set_tests_properties(fu3_qspi_bus PROPERTIES TIMEOUT 30 LABELS "fu3;qspi;integration")
  endif()
endif()
