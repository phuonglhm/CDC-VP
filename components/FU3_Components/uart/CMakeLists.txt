cmake_minimum_required(VERSION 3.16)
project(fu3_uart VERSION 0.1.0 LANGUAGES CXX)
include(CTest)
if(NOT TARGET SystemC::systemc)
  if(SYSTEMC_HOME)
    list(PREPEND CMAKE_PREFIX_PATH "${SYSTEMC_HOME}")
  endif()
  find_package(SystemCLanguage CONFIG REQUIRED)
endif()
add_library(fu3_uart src/uart.cpp)
add_library(fu3::uart ALIAS fu3_uart)
set_target_properties(fu3_uart PROPERTIES EXPORT_NAME uart POSITION_INDEPENDENT_CODE ON WINDOWS_EXPORT_ALL_SYMBOLS ON)
target_compile_features(fu3_uart PUBLIC cxx_std_17)
target_include_directories(fu3_uart PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include> $<INSTALL_INTERFACE:include>)
target_link_libraries(fu3_uart PUBLIC SystemC::systemc)
if(NOT MSVC)
  target_compile_options(fu3_uart PRIVATE -Wall -Wextra -Wpedantic)
endif()
if(BUILD_TESTING)
  if(NOT TARGET fu3::bus_system)
    # Own copy of the bus: skip its tests in this scope only, never in the cache.
    set(FU3_BUS_BUILD_TESTS OFF)
    add_subdirectory(../bus bus)
  endif()
  foreach(kind behavior integration boundaries)
    add_executable(uart_${kind}_tests tests/${kind}.cpp)
    target_link_libraries(uart_${kind}_tests PRIVATE fu3::uart fu3::bus_system)
    add_test(NAME uart_${kind} COMMAND uart_${kind}_tests)
    set_tests_properties(uart_${kind} PROPERTIES TIMEOUT 30)
  endforeach()
  foreach(variant "32;16" "64;8" "32;8")
    list(GET variant 0 depth)
    list(GET variant 1 oversampling)
    add_test(NAME uart_behavior_${depth}_${oversampling} COMMAND uart_behavior_tests ${depth} ${oversampling})
    set_tests_properties(uart_behavior_${depth}_${oversampling} PROPERTIES TIMEOUT 30)
  endforeach()
endif()
install(TARGETS fu3_uart EXPORT fu3-uart-targets ARCHIVE DESTINATION lib LIBRARY DESTINATION lib RUNTIME DESTINATION bin)
install(EXPORT fu3-uart-targets NAMESPACE fu3:: DESTINATION lib/cmake/fu3-uart)
install(FILES cmake/fu3-uart-config.cmake DESTINATION lib/cmake/fu3-uart)
install(DIRECTORY include/ DESTINATION include)
